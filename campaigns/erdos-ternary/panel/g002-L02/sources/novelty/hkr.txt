and Ramaré [5], but is still inspiring further investigation of the central binomial
coefficients. One question left unanswered can be found in Concrete Mathematics
[4] and is the following conjecture, which is the starting point of this paper.

 import java.io.FileWriter;
 import java.io.IOException;
 import java.io.File;

 class NewSearcher {
     private static int[] number = new int[35];
     private static int size = 0;
     private static final int MAX_SIZE = 35;

    private static final String ERROR_FILE = "Check_needed.txt";

    public static void main(String[] args) {
        deleteFile(ERROR_FILE);

        addNum(1);

        for (int a=0; a<10000000; a++) {
            for (int b=0; b<1000000; b++) {
                if (doubleIt()) {
                    String output = String.format("%d%06d", a, b);
                    System.out.println(output);
                    writeNumberToFile(ERROR_FILE, output);
                }
            }
        }
    }

    private static void addNum(int num) {
        if (size < MAX_SIZE) {
            number[size] = num;
            size ++;
        }
    }

    public static boolean doubleIt() {
